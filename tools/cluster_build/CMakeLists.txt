cmake_minimum_required(VERSION 3.24)
project(strata_cluster_builder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(STRATA_GGML_DIR "${STRATA_ROOT}/third_party/llama.cpp" CACHE PATH "Pinned llama.cpp source (JSON and SHA256 only)")
if(NOT EXISTS "${STRATA_GGML_DIR}/vendor/nlohmann/json.hpp")
  message(FATAL_ERROR "Set STRATA_GGML_DIR to the project's pinned llama.cpp checkout")
endif()
add_library(cluster_build_support STATIC
  build.cpp dependencies.cpp
  "${STRATA_ROOT}/src/runtime/cluster_launch.cpp"
  "${STRATA_ROOT}/src/runtime/cluster_config.cpp"
  "${STRATA_GGML_DIR}/vendor/hash/sha256/sha256.c")
target_include_directories(cluster_build_support PUBLIC "${STRATA_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(cluster_build_support SYSTEM PUBLIC "${STRATA_GGML_DIR}/vendor")
target_include_directories(cluster_build_support SYSTEM PRIVATE "${STRATA_GGML_DIR}/vendor/hash/sha256" "${STRATA_GGML_DIR}/vendor/hash")
find_package(Threads REQUIRED)
target_link_libraries(cluster_build_support PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(cluster_build_support PUBLIC advapi32 ws2_32)
endif()
add_executable(strata-cluster-build main.cpp)
target_link_libraries(strata-cluster-build PRIVATE cluster_build_support)
option(STRATA_BUILDER_TESTS "Build packaging tests (no GPU, downloads or Python)" OFF)
if(STRATA_BUILDER_TESTS)
  enable_testing()
  add_executable(cluster-build-test test.cpp)
  target_link_libraries(cluster-build-test PRIVATE cluster_build_support)
  add_test(NAME cluster-build-test COMMAND cluster-build-test)
  set_tests_properties(cluster-build-test PROPERTIES WORKING_DIRECTORY "${STRATA_ROOT}")
endif()
